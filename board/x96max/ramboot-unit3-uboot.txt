# Unit 3 only. This unit's U-Boot prompt is g12a_u212_v1# and its aml_dt is
# sm1_ac213_4g, so the machine is SM1 (S905X3) even though the board id says
# u212: the pick below loads meson-g12a-x96-max-p-sm1.dtb, the DTB every
# successful boot since round 205 used (vdec = amlogic,sm1-vdec; the plain
# meson-g12a-x96-max-p.dtb is the G12A one and breaks USB/vdec here).
# SD is U-Boot mmc0:1. Legacy uImage-wrapped gzip CPIO is required by this
# vendor U-Boot 2015.01. Each attempt gates booti on all three loads: never
# boot stale memory. Vendor console accepts at most 511 bytes per line: keep
# definitions separate.
setenv ram_boot_setup 'setenv kernel_addr_r 0x01080000; setenv fdt_addr_r 0x08000000; setenv dtb_mem_addr 0x08000000; setenv fdtaddr 0x08000000; setenv ramdisk_addr_r 0x10000000'
setenv ram_boot_pick_dtb 'setenv ramboot_dtb meson-g12a-x96-max-p-sm1.dtb; echo RAMBOOT_DTB=${ramboot_dtb}'
setenv ram_boot_load 'if fatload mmc 0:1 ${kernel_addr_r} Image; then if fatload mmc 0:1 ${fdt_addr_r} ${ramboot_dtb}; then if fatload mmc 0:1 ${ramdisk_addr_r} rootfs.cpio.uboot; then setenv bootargs console=ttyAML0,115200n8 earlycon rdinit=/init panic_on_oops=1 panic=8; booti ${kernel_addr_r} ${ramdisk_addr_r} ${fdt_addr_r}; fi; fi; fi'
setenv linux_boot_ram 'run ram_boot_setup; run ram_boot_pick_dtb; for ram_try in 1 2 3; do if usb start; then sleep 3; if mmc dev 0; then if mmc rescan; then run ram_boot_load; fi; fi; fi; done; echo ERROR: RAM boot failed'
# Round 137: without "usb start" in U-Boot the glue is never initialised and
# the USB area aborts inside Linux, so the retry loop above is not optional.
# panic_on_oops=1 panic=8 turns a wedged encoder into a reboot back into this
# same path. Unit 2 runs this exact recipe in production (restart-recipe log).
setenv bootcmd 'run linux_boot_ram'
saveenv
run linux_boot_ram
