# Unit 2 only. All setenv commands below are RAM-only until saveenv.
# Kernel relocation, DTB and initramfs ranges do not overlap firmware/U-Boot.
# Legacy uImage-wrapped gzip CPIO is required by this vendor U-Boot 2015.01.
# Each attempt gates booti on all three loads: never boot stale memory.
# USB cold enumeration needs a settling interval; retry only a bounded 3 times.
# Vendor console accepts at most 511 bytes per line: keep definitions separate.
setenv ram_boot_setup 'setenv kernel_addr_r 0x01080000; setenv fdt_addr_r 0x08000000; setenv dtb_mem_addr 0x08000000; setenv fdtaddr 0x08000000; setenv ramdisk_addr_r 0x10000000'
setenv ram_boot_load 'if fatload usb 0:1 ${kernel_addr_r} Image; then if fatload usb 0:1 ${fdt_addr_r} meson-g12a-x96-max-p-ephy-4g.dtb; then if fatload usb 0:1 ${ramdisk_addr_r} rootfs.cpio.uboot; then setenv bootargs console=ttyAML0,115200n8 earlycon rdinit=/init; booti ${kernel_addr_r} ${ramdisk_addr_r} ${fdt_addr_r}; fi; fi; fi'
setenv linux_boot_ram 'run ram_boot_setup; for ram_try in 1 2 3; do if usb start; then sleep 3; run ram_boot_load; fi; echo RAM boot attempt ${ram_try} failed; if test ${ram_try} != 3; then usb reset; sleep 2; fi; done; echo ERROR: RAM boot failed; false'
run linux_boot_ram
