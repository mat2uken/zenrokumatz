# Unit 1 only: g12a_u212_2g, internal PHY, SD is U-Boot mmc0:1.
# All changes are RAM-only until saveenv. Keep every console line under 500 bytes.
# Three files are loaded before booti; a failed mmc/fatload never boots stale RAM.
setenv ram_boot_setup 'setenv sddev 0; setenv kernel_addr_r 0x01080000; setenv fdt_addr_r 0x08000000; setenv dtb_mem_addr 0x08000000; setenv fdtaddr 0x08000000; setenv ramdisk_addr_r 0x10000000'
setenv linux_boot_ram 'run ram_boot_setup; if mmc dev ${sddev}; then if mmc rescan; then if fatload mmc ${sddev}:1 ${kernel_addr_r} Image; then if fatload mmc ${sddev}:1 ${fdt_addr_r} meson-g12a-x96-max-p.dtb; then if fatload mmc ${sddev}:1 ${ramdisk_addr_r} rootfs.cpio.uboot; then setenv bootargs console=ttyAML0,115200n8 earlycon rdinit=/init; booti ${kernel_addr_r} ${ramdisk_addr_r} ${fdt_addr_r}; fi; fi; fi; fi; fi'
run linux_boot_ram
